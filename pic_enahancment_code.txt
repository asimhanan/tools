import tkinter as tk
from tkinter import filedialog, ttk, messagebox
from PIL import Image, ImageTk, ImageEnhance, ImageFilter
import cv2
import numpy as np
import os
from datetime import datetime

class ImageEnhancementTool:
    def __init__(self, root):
        self.root = root
        self.root.title("AI Image Enhancement Tool")
        self.root.geometry("1200x700")
        
        # Variables
        self.image = None
        self.original_image = None
        self.display_image = None
        self.image_path = ""
        
        # Create GUI
        self.create_widgets()
        self.setup_styles()
        
    def create_widgets(self):
        # Left panel - Controls
        control_frame = tk.Frame(self.root, width=300, bg="#2c3e50")
        control_frame.pack(side=tk.LEFT, fill=tk.Y)
        control_frame.pack_propagate(False)
        
        # Title
        title_label = tk.Label(control_frame, text="Image Enhancer", 
                              font=("Arial", 18, "bold"), bg="#2c3e50", fg="white")
        title_label.pack(pady=20)
        
        # Upload button
        upload_btn = tk.Button(control_frame, text="📁 Upload Image", 
                              command=self.upload_image,
                              font=("Arial", 12), bg="#3498db", fg="white",
                              padx=20, pady=10)
        upload_btn.pack(pady=10)
        
        # Enhancement options
        self.create_sliders(control_frame)
        
        # AI Enhancement buttons
        ai_frame = tk.LabelFrame(control_frame, text="AI Enhancements", 
                                font=("Arial", 12, "bold"), bg="#2c3e50", fg="white")
        ai_frame.pack(pady=20, padx=10, fill=tk.X)
        
        buttons = [
            ("🤖 Super Resolution", self.super_resolution),
            ("✨ Enhance Details", self.enhance_details),
            ("😊 Face Enhancement", self.enhance_faces),
            ("🎨 Color Correction", self.color_correction),
            ("🌙 Low-Light Enhance", self.low_light_enhance)
        ]
        
        for text, command in buttons:
            btn = tk.Button(ai_frame, text=text, command=command,
                          bg="#9b59b6", fg="white", font=("Arial", 10))
            btn.pack(pady=5, padx=10, fill=tk.X)
        
        # Reset and Save buttons
        btn_frame = tk.Frame(control_frame, bg="#2c3e50")
        btn_frame.pack(pady=20)
        
        reset_btn = tk.Button(btn_frame, text="🔄 Reset", 
                             command=self.reset_image,
                             bg="#e74c3c", fg="white", font=("Arial", 12))
        reset_btn.grid(row=0, column=0, padx=5)
        
        save_btn = tk.Button(btn_frame, text="💾 Save", 
                            command=self.save_image,
                            bg="#27ae60", fg="white", font=("Arial", 12))
        save_btn.grid(row=0, column=1, padx=5)
        
        # Status label
        self.status_label = tk.Label(control_frame, text="Upload an image to begin", 
                                    bg="#2c3e50", fg="#ecf0f1", font=("Arial", 10))
        self.status_label.pack(pady=10)
        
        # Right panel - Image display
        display_frame = tk.Frame(self.root, bg="#34495e")
        display_frame.pack(side=tk.RIGHT, expand=True, fill=tk.BOTH)
        
        # Canvas for image display
        self.canvas = tk.Canvas(display_frame, bg="#34495e", highlightthickness=0)
        self.canvas.pack(expand=True, fill=tk.BOTH)
        
        # Image info label
        self.info_label = tk.Label(display_frame, text="", 
                                  bg="#34495e", fg="#bdc3c7", font=("Arial", 10))
        self.info_label.pack(pady=5)
    
    def create_sliders(self, parent):
        # Brightness
        self.brightness_var = tk.DoubleVar(value=1.0)
        self.create_slider(parent, "Brightness", self.brightness_var, self.apply_basic_enhancements)
        
        # Contrast
        self.contrast_var = tk.DoubleVar(value=1.0)
        self.create_slider(parent, "Contrast", self.contrast_var, self.apply_basic_enhancements)
        
        # Sharpness
        self.sharpness_var = tk.DoubleVar(value=1.0)
        self.create_slider(parent, "Sharpness", self.sharpness_var, self.apply_basic_enhancements)
        
        # Saturation
        self.saturation_var = tk.DoubleVar(value=1.0)
        self.create_slider(parent, "Saturation", self.saturation_var, self.apply_basic_enhancements)
        
        # Noise reduction
        self.noise_var = tk.DoubleVar(value=0)
        self.create_slider(parent, "Noise Reduction", self.noise_var, self.apply_noise_reduction)
        
        # Resolution scale (for super resolution)
        self.resolution_var = tk.IntVar(value=100)
        self.create_slider(parent, "Resolution Scale (%)", self.resolution_var, self.apply_resolution_scale, 100, 400)
    
    def create_slider(self, parent, label, variable, command, min_val=0, max_val=2):
        frame = tk.Frame(parent, bg="#2c3e50")
        frame.pack(pady=10, padx=10, fill=tk.X)
        
        tk.Label(frame, text=label, bg="#2c3e50", fg="white", 
                font=("Arial", 10)).pack(anchor="w")
        
        slider = ttk.Scale(frame, from_=min_val, to=max_val, 
                          variable=variable, orient=tk.HORIZONTAL,
                          command=lambda x: command())
        slider.pack(fill=tk.X)
        
        value_label = tk.Label(frame, text=f"{variable.get():.1f}", 
                              bg="#2c3e50", fg="#bdc3c7", font=("Arial", 8))
        value_label.pack(anchor="e")
        
        # Update value label when slider changes
        def update_label(val):
            value_label.config(text=f"{float(val):.1f}")
        
        variable.trace("w", lambda *args: update_label(variable.get()))
    
    def setup_styles(self):
        style = ttk.Style()
        style.configure("TScale", background="#2c3e50")
    
    def upload_image(self):
        file_path = filedialog.askopenfilename(
            filetypes=[("Image files", "*.jpg *.jpeg *.png *.bmp *.tiff")]
        )
        
        if file_path:
            try:
                self.image_path = file_path
                self.original_image = Image.open(file_path)
                self.image = self.original_image.copy()
                
                # Reset sliders
                self.brightness_var.set(1.0)
                self.contrast_var.set(1.0)
                self.sharpness_var.set(1.0)
                self.saturation_var.set(1.0)
                self.noise_var.set(0)
                self.resolution_var.set(100)
                
                self.display_image()
                self.update_status(f"Loaded: {os.path.basename(file_path)}")
            except Exception as e:
                messagebox.showerror("Error", f"Failed to load image: {str(e)}")
    
    def display_image(self):
        if self.image:
            # Resize for display while maintaining aspect ratio
            canvas_width = self.canvas.winfo_width()
            canvas_height = self.canvas.winfo_height()
            
            if canvas_width > 1:  # Canvas has been created
                # Calculate display size
                img_width, img_height = self.image.size
                ratio = min(canvas_width/img_width, canvas_height/img_height) * 0.9
                new_size = (int(img_width * ratio), int(img_height * ratio))
                
                display_img = self.image.copy()
                display_img.thumbnail(new_size, Image.Resampling.LANCZOS)
                
                # Convert for Tkinter
                self.tk_image = ImageTk.PhotoImage(display_img)
                
                # Clear canvas and display image
                self.canvas.delete("all")
                self.canvas.create_image(canvas_width//2, canvas_height//2, 
                                       image=self.tk_image, anchor=tk.CENTER)
                
                # Update info label
                info_text = f"Size: {self.image.size[0]}x{self.image.size[1]} | "
                info_text += f"Mode: {self.image.mode}"
                self.info_label.config(text=info_text)
    
    def apply_basic_enhancements(self):
        if self.image and self.original_image:
            self.image = self.original_image.copy()
            
            # Apply enhancements
            enhancer = ImageEnhance.Brightness(self.image)
            self.image = enhancer.enhance(self.brightness_var.get())
            
            enhancer = ImageEnhance.Contrast(self.image)
            self.image = enhancer.enhance(self.contrast_var.get())
            
            enhancer = ImageEnhance.Sharpness(self.image)
            self.image = enhancer.enhance(self.sharpness_var.get())
            
            enhancer = ImageEnhance.Color(self.image)
            self.image = enhancer.enhance(self.saturation_var.get())
            
            self.display_image()
    
    def apply_noise_reduction(self):
        if self.image:
            try:
                # Convert PIL to OpenCV
                img_cv = cv2.cvtColor(np.array(self.image), cv2.COLOR_RGB2BGR)
                
                # Apply bilateral filter for noise reduction
                denoised = cv2.bilateralFilter(
                    img_cv, 
                    9, 
                    self.noise_var.get() * 20, 
                    self.noise_var.get() * 20
                )
                
                # Convert back to PIL
                self.image = Image.fromarray(cv2.cvtColor(denoised, cv2.COLOR_BGR2RGB))
                self.display_image()
            except:
                self.apply_basic_enhancements()  # Fallback
    
    def apply_resolution_scale(self):
        if self.image and self.original_image:
            scale = self.resolution_var.get() / 100.0
            
            if scale != 1.0:
                new_size = (int(self.original_image.size[0] * scale),
                           int(self.original_image.size[1] * scale))
                
                # Use high-quality resampling
                self.image = self.original_image.resize(new_size, Image.Resampling.LANCZOS)
                self.apply_basic_enhancements()
                self.update_status(f"Resolution scaled to {new_size[0]}x{new_size[1]}")
    
    def super_resolution(self):
        if not self.image:
            messagebox.showwarning("No Image", "Please upload an image first")
            return
        
        try:
            # Simple super-resolution using upscaling with sharpening
            messagebox.showinfo("Super Resolution", 
                              "Enhancing resolution using AI algorithm...")
            
            # Upscale by 2x using LANCZOS
            width, height = self.image.size
            new_size = (width * 2, height * 2)
            
            # Enhanced upscaling with multiple passes
            temp = self.image.resize(new_size, Image.Resampling.LANCZOS)
            
            # Apply edge enhancement
            temp = temp.filter(ImageFilter.UnsharpMask(radius=2, percent=150, threshold=3))
            
            self.image = temp
            self.display_image()
            self.update_status("Super resolution applied (2x upscale)")
            
        except Exception as e:
            messagebox.showerror("Error", f"Super resolution failed: {str(e)}")
    
    def enhance_details(self):
        if self.image:
            try:
                # Apply multiple filters for detail enhancement
                self.image = self.image.filter(ImageFilter.DETAIL)
                self.image = self.image.filter(ImageFilter.EDGE_ENHANCE_MORE)
                
                # Apply unsharp mask for clarity
                self.image = self.image.filter(
                    ImageFilter.UnsharpMask(radius=2, percent=100, threshold=0)
                )
                
                self.display_image()
                self.update_status("Details enhanced")
            except:
                self.apply_basic_enhancements()
    
    def enhance_faces(self):
        messagebox.showinfo("Face Enhancement", 
                          "Face enhancement requires additional face detection libraries.\n\n"
                          "To enable this feature, install:\n"
                          "pip install opencv-contrib-python")
    
    def color_correction(self):
        if self.image:
            try:
                # Convert to numpy array for OpenCV processing
                img_array = np.array(self.image)
                
                # Convert to LAB color space
                lab = cv2.cvtColor(img_array, cv2.COLOR_RGB2LAB)
                
                # Split LAB channels
                l, a, b = cv2.split(lab)
                
                # Apply CLAHE to L channel for better contrast
                clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8,8))
                l = clahe.apply(l)
                
                # Merge channels and convert back to RGB
                enhanced = cv2.merge([l, a, b])
                enhanced = cv2.cvtColor(enhanced, cv2.COLOR_LAB2RGB)
                
                self.image = Image.fromarray(enhanced)
                self.display_image()
                self.update_status("Color correction applied")
            except:
                self.apply_basic_enhancements()
    
    def low_light_enhance(self):
        if self.image:
            try:
                # Convert to numpy array
                img_array = np.array(self.image).astype('float32') / 255.0
                
                # Apply gamma correction for low-light enhancement
                gamma = 0.5  # Darken first
                img_array = img_array ** gamma
                
                # Increase contrast
                img_array = cv2.normalize(img_array, None, 0, 1, cv2.NORM_MINMAX)
                
                # Convert back to PIL
                img_array = (img_array * 255).astype('uint8')
                self.image = Image.fromarray(img_array)
                
                self.display_image()
                self.update_status("Low-light enhancement applied")
            except:
                self.apply_basic_enhancements()
    
    def reset_image(self):
        if self.original_image:
            self.image = self.original_image.copy()
            self.brightness_var.set(1.0)
            self.contrast_var.set(1.0)
            self.sharpness_var.set(1.0)
            self.saturation_var.set(1.0)
            self.noise_var.set(0)
            self.resolution_var.set(100)
            self.display_image()
            self.update_status("Image reset to original")
    
    def save_image(self):
        if self.image:
            file_path = filedialog.asksaveasfilename(
                defaultextension=".png",
                filetypes=[("PNG files", "*.png"), 
                          ("JPEG files", "*.jpg"), 
                          ("All files", "*.*")]
            )
            
            if file_path:
                try:
                    self.image.save(file_path, quality=95)
                    self.update_status(f"Image saved: {os.path.basename(file_path)}")
                    messagebox.showinfo("Success", "Image saved successfully!")
                except Exception as e:
                    messagebox.showerror("Error", f"Failed to save image: {str(e)}")
    
    def update_status(self, message):
        self.status_label.config(text=message)
        self.root.after(3000, lambda: self.status_label.config(
            text=f"Ready | {os.path.basename(self.image_path) if self.image_path else 'No image loaded'}"
        ))
    
    def on_resize(self, event):
        self.display_image()

if __name__ == "__main__":
